# 测试 2：strict 行为差异测试（PowerShell 版，修复中文编码）
# 用法：拷贝本脚本到内网任意目录，设置环境变量后执行

param(
    [string]$GatewayUrl = $env:GATEWAY_URL,
    [string]$ApiKey     = $env:API_KEY,
    [string]$Model      = $env:MODEL
)

if (-not $GatewayUrl -or -not $ApiKey -or -not $Model) {
    Write-Error "缺少必要参数。请设置环境变量 GATEWAY_URL、API_KEY、MODEL。"
    exit 1
}

# 设置控制台编码为 UTF-8，避免 curl 输出被 PowerShell 用 GBK 解码
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8
$PSDefaultParameterValues['Out-File:Encoding'] = 'utf8'

$PROMPT  = '请帮我看看当前目录下有哪些txt文件'
$OutDir  = if ($PSScriptRoot) { $PSScriptRoot } else { "." }

foreach ($STRICT in @($true, $false)) {
    $strictStr = $STRICT.ToString().ToLower()
    Write-Host "=== strict: $strictStr ===" -ForegroundColor Cyan

    $jsonBody = @{
        model    = $Model
        messages = @(@{ role = "user"; content = $PROMPT })
        tools    = @(@{
            type     = "function"
            function = @{
                name                 = "Grep"
                strict               = $STRICT
                description          = "Search file contents using regex"
                parameters           = @{
                    type                 = "object"
                    properties           = @{
                        pattern = @{ type = "string" }
                        path    = @{ type = "string" }
                        glob    = @{ type = "string" }
                    }
                    required             = @("pattern", "path", "glob")
                    additionalProperties = $false
                }
            }
        })
        tool_choice = "auto"
    } | ConvertTo-Json -Depth 10

    # 保存请求 JSON
    $reqFile = Join-Path $OutDir "test_strict_${strictStr}.json"
    [System.IO.File]::WriteAllText($reqFile, $jsonBody, [System.Text.UTF8Encoding]::new($false))

    $MISSING = 0
    for ($i = 1; $i -le 10; $i++) {
        $respFile = Join-Path $OutDir "tmp_resp_${strictStr}_${i}.json"

        # curl 直接输出到文件，绕过 PowerShell 的编码问题
        $null = curl.exe -sS -X POST "$GatewayUrl/chat/completions" `
            -H "Authorization: Bearer $ApiKey" `
            -H "Content-Type: application/json" `
            -d "@$reqFile" `
            -o "$respFile"

        try {
            # 用 UTF-8 读取文件
            $respText = [System.IO.File]::ReadAllText($respFile, [System.Text.Encoding]::UTF8)
            $obj = $respText | ConvertFrom-Json

            $toolCalls = $obj.choices[0].message.tool_calls
            if ($toolCalls -and $toolCalls.Count -gt 0) {
                $argsStr = $toolCalls[0].function.arguments
                if ($argsStr -and ($argsStr -notmatch '"pattern"')) {
                    $MISSING++
                    Write-Host "  Run #$i 漏字段: $argsStr" -ForegroundColor Yellow
                } else {
                    # Write-Host "  Run #$i 正常: $argsStr" -ForegroundColor DarkGray
                }
            } else {
                Write-Host "  Run #$i 无 tool_calls" -ForegroundColor DarkGray
            }
        } catch {
            Write-Host "  Run #$i 解析失败: $_" -ForegroundColor Red
            # 把原始内容打印出来方便排查
            $raw = [System.IO.File]::ReadAllText($respFile, [System.Text.Encoding]::UTF8)
            Write-Host "  原始响应前500字符: $($raw.Substring(0, [Math]::Min(500, $raw.Length)))" -ForegroundColor DarkGray
        } finally {
            # 清理临时文件
            if (Test-Path $respFile) { Remove-Item $respFile -Force }
        }
    }

    Write-Host "  $strictStr 模式：10 次中 $MISSING 次漏 pattern" -ForegroundColor Green
    Write-Host ""
}
